name: Generate project pages

on:
  workflow_run:
    workflows: ["Update visitor count"] # Must match the exact 'name:' of the generator workflow
    types:
      - completed
  push:
    branches: [main]
  # index.html changing isn't the only thing that can go stale - a
  # linked repo's README can update on its own, so re-run periodically
  # too. Remove this if you'd rather it only run on push.
  schedule:
    - cron: "5 0 * * *"  # every day, 00:00 UTC
  workflow_dispatch: {}

# Shared with update-visitor-count.yml's concurrency group: this workflow
# triggers on every push update-visitor-count.yml makes (workflow_run
# below, plus push:branches:[main] above), so the two raced to commit+
# push at nearly the same time on almost every run -- see the comment
# in update-visitor-count.yml for the full story. Same group name on
# both means GitHub queues them against each other instead of letting
# both write to main concurrently.
concurrency:
  group: main-branch-writes
  cancel-in-progress: false

permissions:
  contents: write  # git-auto-commit-action needs push rights

jobs:
  generate:
    if: ${{ github.event_name != 'workflow_run' || github.event.workflow_run.conclusion == 'success' }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install dependencies
        run: pip install requests beautifulsoup4 pygments

      - name: Generate project pages
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: python .github/scripts/generate_project_pages.py

      - name: Commit generated pages
        uses: stefanzweifel/git-auto-commit-action@v5
        with:
          commit_message: "chore: regenerate project pages"
          # .github/data/project-pages-cache.json is the change-skip
          # build cache (see generate_project_pages.py's module
          # docstring) - it has to be committed back too, or every run
          # starts from an empty cache and re-fetches everything from
          # GitHub again, defeating the whole point of it.
          file_pattern: "projects/*.html .github/data/project-pages-cache.json"
