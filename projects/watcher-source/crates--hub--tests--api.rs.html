const CONFIG: &str = r#"
[check.http]
command = "check_http -H $host.address$"
interval = "1h"

[service.http]

[hostgroup.web]
services = ["http"]

[[host]]
name = "web01"
groups = ["web"]

[[host]]
name = "db01"
host_check = "backup"
  [host.service.backup]
  passive = true
  [host.service.dump]
  passive = true
"#;

fn config() -> Arc<Config> {
    config_from(CONFIG)
}

fn config_from(text: &str) -> Arc<Config> {
    let dir: PathBuf = std::env::temp_dir().join(format!(
        "watcher-hub-{}-{:?}",
        std::process::id(),
        std::thread::current().id()
    ));
    std::fs::create_dir_all(&dir).unwrap();
    std::fs::write(dir.join("conf.toml"), text).unwrap();
    let loaded = load_dir(&dir).unwrap();
    std::fs::remove_dir_all(&dir).unwrap();
    Arc::new(loaded.config)
}

/// A hub with a running scheduler (its active check runs only hourly).
fn hub() -> Arc<Hub> {
    hub_with(config())
}

fn hub_with(config: Arc<Config>) -> Arc<Hub> {
    let (control, control_rx) = mpsc::channel(8);
    let (events_tx, mut events) = mpsc::channel::<Event>(64);
    let hub = Hub::new(
        config.clone(),
        HashMap::new(),
        control,
        None,
        AuthSettings::OFF,
    );
    tokio::spawn(schedule(
        jobs(&config, &CheckDirs::default()),
        HashMap::new(),
        4,
        Arc::new(watcher_exec::ssh::Pool::unavailable("no SSH in tests")),
        control_rx,
        events_tx,
    ));
    let recorder = hub.clone();
    tokio::spawn(async move {
        while let Some(event) = events.recv().await {
            recorder.record(&event);
        }
    });
    hub
}

async fn call(hub: &Arc<Hub>, method: &str, uri: &str, body: Option<Value>) -> (StatusCode, Value) {
    let mut req = Request::builder().method(method).uri(uri);
    let body = match body {
        Some(json) => {
            req = req.header("content-type", "application/json");
            Body::from(json.to_string())
        }
        None => Body::empty(),
    };
    let resp = hub
        .clone()
        .router()
        .oneshot(req.body(body).unwrap())
        .await
        .unwrap();
    let status = resp.status();
    let bytes = resp.into_body().collect().await.unwrap().to_bytes();
    let json = serde_json::from_slice(&bytes).unwrap_or(Value::Null);
    (status, json)
}

fn event(id: &str, exit: i32, stdout: &str) -> Event {
    let raw = RawOutput {
        termination: Termination::Exited(exit),
        stdout: stdout.into(),
        stderr: String::new(),
    };
    let result = evaluate(&raw, Format::Auto, &Default::default());
    let (state, change) = ServiceState::first(result.state, 3, false, SystemTime::now());
    Event {
        id: id.into(),
        result,
        duration: Duration::from_millis(5),
        state,
        change: Some(change),
        flap: None,
    }
}

#[tokio::test]
async fn health_and_lists() {
    let hub = hub();
    let (status, health) = call(&hub, "GET", "/api/v1/health", None).await;
    assert_eq!(status, StatusCode::OK);
    // db01's host check is the host's, not one of its services.
    assert_eq!(health["services"], 2);
    assert_eq!(health["database"], Value::Null);

    let (_, list) = call(&hub, "GET", "/api/v1/services", None).await;
    assert_eq!(list["counts"]["pending"], 2);
    assert_eq!(list["services"][0]["state"], Value::Null);

    hub.record(&event("web01/http", 2, "HTTP CRITICAL - down | time=1.5s"));
    let (_, problems) = call(&hub, "GET", "/api/v1/services?problems=true", None).await;
    assert_eq!(problems["services"].as_array().unwrap().len(), 1);
    assert_eq!(problems["services"][0]["state"], "critical");
    assert_eq!(problems["services"][0]["state_type"], "soft");
    let (_, hard) = call(
        &hub,
        "GET",
        "/api/v1/services?problems=true&hard=true",
        None,
    )
    .await;
    assert_eq!(hard["services"].as_array().unwrap().len(), 0);
    let (_, by_state) = call(&hub, "GET", "/api/v1/services?state=pending", None).await;
    assert_eq!(by_state["services"][0]["service"], "dump");
    let (status, _) = call(&hub, "GET", "/api/v1/services?state=bad", None).await;
    assert_eq!(status, StatusCode::BAD_REQUEST);

    let (_, list) = call(&hub, "GET", "/api/v1/hosts", None).await;
    assert_eq!(list["total"], 2);
    let hosts = &list["hosts"];
    assert_eq!(hosts[0]["name"], "web01");
    // No host check: UP. The worst service state is separate.
    assert_eq!(hosts[0]["state"], "up");
    assert_eq!(hosts[0]["check"], Value::Null);
    assert_eq!(hosts[0]["worst"], "critical");
    // The host check has no result yet: pending.
    assert_eq!(hosts[1]["state"], Value::Null);
    assert_eq!(hosts[1]["check"], "backup");
    assert_eq!(hosts[1]["worst"], Value::Null);

    let (status, web) = call(&hub, "GET", "/api/v1/hosts/web01", None).await;
    assert_eq!(status, StatusCode::OK);
    assert_eq!(web["services"][0]["output"], "HTTP CRITICAL - down");
    assert_eq!(web["check"], Value::Null);
    let (_, db) = call(&hub, "GET", "/api/v1/hosts/db01", None).await;
    assert_eq!(db["check"]["service"], "backup");
    assert_eq!(db["services"].as_array().unwrap().len(), 1);
    assert_eq!(db["services"][0]["service"], "dump");
    let (status, err) = call(&hub, "GET", "/api/v1/hosts/nope", None).await;
    assert_eq!(status, StatusCode::NOT_FOUND);
    assert_eq!(err["error"], "no host \"nope\"");

    let (_, detail) = call(&hub, "GET", "/api/v1/services/web01/http", None).await;
    assert_eq!(detail["metrics"][0]["name"], "time");
    assert_eq!(detail["metrics"][0]["value"], 1.5);
    assert_eq!(detail["check"]["command"], "check_http -H web01");
    assert_eq!(detail["check"]["interval_s"], 3600.0);
    assert_eq!(detail["duration_ms"], 5.0);
}

#[tokio::test]
async fn passive_results() {
    let hub = hub();
    let body = json!({"exit_code": 2, "output": "BACKUP CRITICAL - failed | size=0B"});
    let (status, detail) = call(
        &hub,
        "POST",
        "/api/v1/services/db01/backup/result",
        Some(body),
    )
    .await;
    assert_eq!(status, StatusCode::OK, "{detail}");
    assert_eq!(detail["state"], "critical");
    assert_eq!(detail["output"], "BACKUP CRITICAL - failed");
    assert_eq!(detail["passive"], true);
    assert_eq!(detail["check"]["command"], Value::Null);
    assert_eq!(detail["host_check"], true);

    // It is db01's host check: the host is DOWN (passive: 1 attempt, hard).
    let (_, db) = call(&hub, "GET", "/api/v1/hosts/db01", None).await;
    assert_eq!(db["host"]["state"], "down");
    assert_eq!(db["host"]["state_type"], "hard");
    assert_eq!(db["host"]["output"], "BACKUP CRITICAL - failed");
    // Other services of the host don't notify; the host check does.
    assert!(hub.host_down("db01/dump"));
    assert!(!hub.host_down("db01/backup"));
    assert!(!hub.host_down("web01/http"));

    // An open problem for repeats; held back once acknowledged.
    let open = hub.open_problems();
    assert_eq!(
        open.iter()
            .map(|p| (p.id.as_str(), p.suppressed.as_deref()))
            .collect::<Vec<_>>(),
        [("db01/backup", None)]
    );
    assert_eq!(open[0].output, "BACKUP CRITICAL - failed");
    let (status, ack) = call(
        &hub,
        "POST",
        "/api/v1/actions/ack",
        Some(json!({ "hosts": ["db01"] })),
    )
    .await;
    assert_eq!(status, StatusCode::OK, "{ack}");
    assert_eq!(
        hub.open_problems()[0].suppressed.as_deref(),
        Some("acknowledged")
    );

    let body = json!({"state": "ok", "output": "fine"});

#[tokio::test]
async fn dependencies_hold_notifications_back() {
    let hub = hub_with(config_from(&format!(
        r#"{CONFIG}
[[dependency]]
name = "web-needs-db"
on = "db01"
hosts = ["web01"]

[[dependency]]
name = "web-needs-dump"
on = "db01/dump"
services = {{ name = ["http"] }}
"#
    )));
    let critical = watcher_core::State::Critical;
    assert_eq!(hub.suppressed("web01/http", critical), None);

    // The master service goes CRITICAL (passive: 1 attempt, hard).
    let body = json!({"exit_code": 2, "output": "DUMP CRITICAL - too old"});
    call(
        &hub,
        "POST",
        "/api/v1/services/db01/dump/result",
        Some(body),
    )
    .await;
    assert_eq!(
        hub.suppressed("web01/http", critical).as_deref(),
        Some("depends on db01/dump (CRITICAL)")
    );
    // The master's own notification is not held back.
    assert_eq!(hub.suppressed("db01/dump", critical), None);

    // WARNING is not in the default states: nothing is held back.
    let body = json!({"exit_code": 1, "output": "DUMP WARNING - old"});
    call(
        &hub,
        "POST",
        "/api/v1/services/db01/dump/result",
        Some(body),
    )
    .await;
    assert_eq!(hub.suppressed("web01/http", critical), None);

    // The master host goes DOWN: the dependent host's services wait too.
    let body = json!({"exit_code": 2, "output": "BACKUP CRITICAL - failed"});
    call(
        &hub,
        "POST",
        "/api/v1/services/db01/backup/result",
        Some(body),
    )
    .await;
    assert_eq!(
        hub.suppressed("web01/http", critical).as_deref(),
        Some("depends on db01 (DOWN)")
    );
    // Its own host being DOWN still comes first, and the host check (the
    // master's own alert) is never held back.
    assert_eq!(
        hub.suppressed("db01/dump", critical).as_deref(),
        Some("host is DOWN")
    );
    assert_eq!(hub.suppressed("db01/backup", critical), None);

    // The master recovers: notifications flow again.
    let body = json!({"state": "ok", "output": "fine"});
    call(
        &hub,
        "POST",
        "/api/v1/services/db01/backup/result",
        Some(body),
    )
    .await;
    assert_eq!(hub.suppressed("web01/http", critical), None);
}

#[tokio::test]
async fn stream_sends_results() {
    let hub = hub();
    let resp = hub
        .clone()
        .router()
        .oneshot(Request::get("/api/v1/stream").body(Body::empty()).unwrap())
        .await
        .unwrap();
    assert_eq!(resp.headers()["content-type"], "text/event-stream");
    let mut body = resp.into_body();
    // A summary first, then results.
    let first = next_event(&mut body).await;
    assert!(first.starts_with("event: summary\ndata: {"), "{first}");
    assert!(first.contains(r#""pending":2"#), "{first}");

    hub.record(&event("web01/http", 0, "HTTP OK"));
    let text = next_event(&mut body).await;
    assert!(text.starts_with("event: result\ndata: {"), "{text}");
    assert!(text.contains(r#""output":"HTTP OK""#), "{text}");
    // The state changed: a new summary within a second.
    let text = next_event(&mut body).await;
    assert!(text.starts_with("event: summary\n"), "{text}");
    assert!(text.contains(r#""ok":1"#), "{text}");
}

/// The next server-sent event, skipping keep-alive comments.
async fn next_event(body: &mut Body) -> String {
    loop {
        let frame = tokio::time::timeout(Duration::from_secs(5), body.frame())
            .await
            .expect("an event within 5 s")
            .unwrap()
            .unwrap();
        let text = String::from_utf8(frame.into_data().unwrap().to_vec()).unwrap();
        if !text.starts_with(':') {
            return text;
        }
    }
}

#[tokio::test]
async fn lists_filter_sort_and_page() {
    let hub = hub();
    hub.record(&event("web01/http", 2, "HTTP CRITICAL - down"));
    hub.record(&event("db01/dump", 1, "DUMP WARNING - slow"));

    let get = |uri: &'static str| {
        let hub = hub.clone();
        async move {
            let (status, body) = call(&hub, "GET", uri, None).await;
            assert_eq!(status, StatusCode::OK, "{uri}: {body}");
            body
        }
    };
    let names = |list: &Value| -> Vec<String> {
        list["services"]
            .as_array()
            .unwrap()
            .iter()
            .map(|s| {
                format!(
                    "{}/{}",
                    s["host"].as_str().unwrap(),
                    s["service"].as_str().unwrap()
                )
            })
            .collect()
    };

    let all = get("/api/v1/services?sort=status&desc=true").await;
    assert_eq!(names(&all), ["web01/http", "db01/dump"]);
    let page = get("/api/v1/services?sort=status&desc=true&offset=1&limit=1").await;
    assert_eq!(page["total"], 2);
    assert_eq!(names(&page), ["db01/dump"]);
    let by_host = get("/api/v1/services?sort=host").await;
    assert_eq!(names(&by_host), ["db01/dump", "web01/http"]);

    // Counts ignore the state filter, so totals boxes stay complete.
    let warning = get("/api/v1/services?state=warning").await;
    assert_eq!(names(&warning), ["db01/dump"]);
    assert_eq!(warning["counts"]["critical"], 1);
    assert_eq!(warning["counts"]["pending"], 0);
    assert_eq!(warning["total"], 1);

    assert_eq!(names(&get("/api/v1/services?q=SLOW").await), ["db01/dump"]);
    assert_eq!(
        names(&get("/api/v1/services?host=web01").await),
        ["web01/http"]
    );
    assert_eq!(
        names(&get("/api/v1/services?hostgroup=web").await),
        ["web01/http"]
    );
    assert_eq!(
        names(&get("/api/v1/services?host=nope").await),
        Vec::<String>::new()
    );
    let unhandled = get("/api/v1/services?unhandled=true&sort=service").await;
    assert_eq!(names(&unhandled), ["db01/dump", "web01/http"]);

    // An acknowledged problem is handled.
    let body = json!({ "services": ["web01/http"], "comment": "known" });
    let (status, r) = call(&hub, "POST", "/api/v1/actions/ack", Some(body)).await;
    assert_eq!(status, StatusCode::OK, "{r}");
    assert_eq!(
        names(&get("/api/v1/services?unhandled=true").await),
        ["db01/dump"]
    );
    let summary = get("/api/v1/summary").await;
    assert_eq!(summary["services"]["unhandled"], 1);
    assert_eq!(summary["services"]["total"], 2);
    assert_eq!(summary["services"]["critical"], 1);
    assert_eq!(summary["hosts"]["total"], 2);
    assert_eq!(summary["hosts"]["up"], 1);
    assert_eq!(summary["hosts"]["pending"], 1);

    let (status, _) = call(&hub, "GET", "/api/v1/services?sort=bad", None).await;
    assert_eq!(status, StatusCode::BAD_REQUEST);

    let hosts = get("/api/v1/hosts?sort=status&desc=true").await;
    assert_eq!(hosts["hosts"][0]["name"], "db01");
    let up = get("/api/v1/hosts?state=up&q=WEB").await;
    assert_eq!(up["total"], 1);
    assert_eq!(up["counts"]["up"], 1);
    assert_eq!(up["hosts"][0]["services"]["critical"], 1);
    let (status, _) = call(&hub, "GET", "/api/v1/hosts?state=ok", None).await;
    assert_eq!(status, StatusCode::BAD_REQUEST);

    let groups = get("/api/v1/groups").await;
    assert_eq!(groups["hostgroups"][0]["name"], "web");
    assert_eq!(groups["hostgroups"][0]["hosts"]["up"], 1);
    assert_eq!(groups["hostgroups"][0]["services"]["critical"], 1);

    let names = get("/api/v1/names").await;
    assert_eq!(names["hosts"], json!(["web01", "db01"]));
    assert_eq!(names["services"], json!(["dump", "http"]));
}

    assert_eq!(
        hub.suppressed("web01/http", watcher_core::State::Critical)
            .as_deref(),
        Some("acknowledged")
    );
    // OK again: the acknowledgement ends, the recovery notifies.
    hub.record(&event("web01/http", 0, "HTTP OK"));
    let (_, svc) = call(&hub, "GET", "/api/v1/services/web01/http", None).await;
    assert_eq!(svc["acknowledged"], false);
    assert_eq!(hub.suppressed("web01/http", watcher_core::State::Ok), None);

    // Mute a host: its services are muted.
    let (_, mute) = call(
        &hub,
        "POST",
        "/api/v1/actions/mute",
        Some(json!({ "hosts": ["db01"] })),
    )
    .await;
    assert_eq!(mute["done"], json!(["db01"]));
    let (_, list) = call(&hub, "GET", "/api/v1/services?host=db01", None).await;
    assert!(
        list["services"]
            .as_array()
            .unwrap()
            .iter()
            .all(|s| s["muted"] == true)
    );
    assert_eq!(
        hub.suppressed("db01/dump", watcher_core::State::Critical)
            .as_deref(),
        Some("muted")
    );
    let (_, unmute) = call(
        &hub,
        "POST",
        "/api/v1/actions/unmute",
        Some(json!({ "hosts": ["db01"] })),
    )
    .await;
    assert_eq!(unmute["done"], json!(["db01"]));

    // A downtime for an hour, by duration.
    let (status, dt) = call(
        &hub,
        "POST",
        "/api/v1/actions/downtime",
        Some(json!({ "hosts": ["web01"], "duration": "1h", "comment": "patching" })),
    )
    .await;
    assert_eq!(status, StatusCode::OK, "{dt}");
    let id = dt["downtimes"][0]["id"].as_i64().unwrap();
    let (_, list) = call(&hub, "GET", "/api/v1/downtimes", None).await;
    assert_eq!(list[0]["active"], true);
    assert_eq!(list[0]["services"], 1);
    let (_, hosts) = call(&hub, "GET", "/api/v1/hosts", None).await;
    let web = hosts["hosts"]
        .as_array()
        .unwrap()
        .iter()
        .find(|h| h["name"] == "web01")
        .unwrap();
    assert_eq!(web["downtime"], true);
    assert!(
        hub.suppressed("web01/http", watcher_core::State::Critical)
            .unwrap()
            .starts_with("in downtime")
    );
    let (status, _) = call(&hub, "DELETE", &format!("/api/v1/downtimes/{id}"), None).await;
    assert_eq!(status, StatusCode::OK);
    let (_, list) = call(&hub, "GET", "/api/v1/downtimes", None).await;
    assert_eq!(list, json!([]));

    // Errors.
    let (status, err) = call(
        &hub,
        "POST",
        "/api/v1/actions/mute",
        Some(json!({ "hosts": ["nope"] })),
    )
    .await;
    assert_eq!(status, StatusCode::NOT_FOUND, "{err}");
    let (status, _) = call(
        &hub,
        "POST",
        "/api/v1/actions/explode",
        Some(json!({ "hosts": ["web01"] })),
    )
    .await;
    assert_eq!(status, StatusCode::NOT_FOUND);
    let (status, _) = call(
        &hub,
        "POST",
        "/api/v1/actions/downtime",
