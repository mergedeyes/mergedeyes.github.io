// Form fields per object type, and the conversion between form values
// and the fields written to the TOML file. Fields the form doesn't know
// are kept as they are.

import type { Kind } from './api.ts';

export type FieldType =
	| 'text'
	| 'mono'
	| 'list'
	| 'bool'
	| 'int'
	| 'duration'
	| 'select'
	| 'vars'
	| 'thresholds'
	| 'selector'
	| 'services'
	| 'windows';

export interface Field {
	key: string;
	label: string;
	type: FieldType;
	help?: string;
	placeholder?: string;
	options?: string[];
	required?: boolean;
	/** For `bool`: the value when the key is absent. */
	default?: boolean;
	/** Suggestions for `list` fields: names of other objects. */
	suggest?: Kind;
	/** Suggestions for a single name. */
	suggestOne?: Kind;
	/** Suggest the scripts in the check directories. */
	scripts?: boolean;
	/** For `selector`: what it selects. */
	selects?: 'host' | 'service';
	/** For `table.key`: the table may also be written as just this value (`ntfy = "https://…"`). */
	short?: boolean;
	/** Fields that rule this one out: it is greyed out while one of them has a value. */
	exclusive?: string[];
}

export interface Group {
	title: string;
	fields: Field[];
	/**
	 * Settings only some objects have (Proxmox VE): the section shows when
	 * the object has them, or once added with its button or `?type=`.
	 */
	extra?: string;
}

const NAME: Field = { key: 'name', label: 'Name', type: 'text', required: true };
const TITLE: Field = { key: 'title', label: 'Title', type: 'text', placeholder: 'display name' };
const VARS: Field = { key: 'vars', label: 'Variables', type: 'vars', help: '$name$ in commands' };
const ENABLED: Field = { key: 'enabled', label: 'Enabled', type: 'bool', default: true };

const SCHEDULE: Field[] = [
	{ key: 'interval', label: 'Check Interval', type: 'duration', placeholder: '1m' },
	{ key: 'retry_interval', label: 'Retry Interval', type: 'duration', placeholder: 'min(interval, 1m)' },
	{ key: 'max_attempts', label: 'Max Attempts', type: 'int', placeholder: '3 (passive: 1)' },
	{ key: 'timeout', label: 'Timeout', type: 'duration', placeholder: '60s' }
];

const COMMAND: Field = { key: 'command', label: 'Command', type: 'mono', scripts: true, placeholder: 'check_ping -H $host.address$ -w $warn$ -c $crit$' };
const CHECK: Field[] = [
	{ ...COMMAND, exclusive: ['builtin'], help: 'or a built-in check' },
	{
		key: 'builtin',
		label: 'Built-in',
		type: 'select',
		placeholder: 'none',
		exclusive: ['command'],
		help: 'a check built into Watcher instead of a command; its options appear below'
	},
	{ key: 'shell', label: 'Shell', type: 'bool', default: false, help: 'run with /bin/sh -c' },
	{ key: 'run_on', label: 'Run On', type: 'select', options: ['local', 'host'], placeholder: 'local', help: 'host: over SSH, see the host’s SSH settings' },
	{ key: 'copy', label: 'Copy', type: 'bool', default: false, help: 'upload the script from the check directories to the host first' },
	{ key: 'passive', label: 'Passive', type: 'bool', default: false, help: 'results via API, no command' },
	{ key: 'format', label: 'Format', type: 'select', options: ['auto', 'nagios', 'key_value', 'json'] }
];

const PASSIVE: Field[] = [
	{ key: 'freshness', label: 'Freshness', type: 'duration', placeholder: 'passive only, e.g. 26h' },
	{ key: 'flap_detection', label: 'Flap Detection', type: 'bool', default: true },
	{ key: 'flap_high', label: 'Flap Start %', type: 'int', placeholder: '20' },
	{ key: 'flap_low', label: 'Flap Stop %', type: 'int', placeholder: '5' }
];

const THRESHOLDS: Field = { key: 'thresholds', label: 'Thresholds', type: 'thresholds', help: '> 100, >= 10GB, 10:20, @10:20' };
const GRAPHS: Field = { key: 'graphs', label: 'Graphs', type: 'bool', default: true, help: 'graphs of its metrics' };
const SERVICES: Field = { key: 'services', label: 'Services', type: 'list', suggest: 'service', help: 'comma-separated' };
const SERVICEGROUPS: Field = {

};

const GRAPH: Field = { key: 'graph', label: 'Graph', type: 'text', help: 'host page graphs; default, or if the host lacks it: Host Check' };
const SSH: Field[] = [

/** Settings of one service on a host (`[host.service.<name>]`). */
export interface ServiceRow {
	name: string;
	/** `yes`, `no`, or `` for the service's own. */
	enabled: string;
	interval: string;
	vars: [string, string][];
	thresholds: [string, string, string][];
	/** Its key in the file; `` for a row the form added. */
	orig: string;
	/** Written even without settings: a service only this entry puts on the host. */
	own: boolean;
}

/** A schedule window row: days (comma-separated), from, to. */
export type WindowRow = [string, string, string];

/** Form state: strings for inputs, rows for vars, thresholds, services and windows, a map for selectors. */
export type FormValue =
	| string
	| boolean
	| null
	| [string, string][]
	| [string, string, string][]
	| ServiceRow[]
	| WindowRow[]
	| Record<string, string>;
export type FormValues = Record<string, FormValue>;

function fields(kind: Kind): Field[] {
	return SCHEMAS[kind].flatMap((g) => g.fields);
}

function str(v: unknown): string {
	return v === undefined || v === null ? '' : String(v);
}

function list(v: unknown): string[] {
	if (v === undefined || v === null) return [];
	return Array.isArray(v) ? v.map(String) : [String(v)];
}

function items(text: string): string[] {
	return text
		.split(',')
		.map((s) => s.trim())
		.filter(Boolean);
}

/** A deep copy of TOML data; works on Svelte state proxies, unlike structuredClone. */
function copy<T>(v: T): T {
	return JSON.parse(JSON.stringify(v)) as T;
}

function isTable(v: unknown): v is Record<string, unknown> {
	return typeof v === 'object' && v !== null && !Array.isArray(v);
}

/** The value at a field key, dotted for nested tables (`ssh.user`). */
function at(raw: Record<string, unknown>, key: string): unknown {
	let v: unknown = raw;
	for (const part of key.split('.')) v = isTable(v) ? v[part] : undefined;
	return v;
}

/** A field's value; a `short` field also reads the table written as just its value. */
function value(raw: Record<string, unknown>, f: Field): unknown {
	const head = f.key.split('.')[0]!;
	return f.short && typeof raw[head] === 'string' ? raw[head] : at(raw, f.key);
}

/** Sets a field key, or removes it (`undefined`) with tables left empty. */
function put(out: Record<string, unknown>, key: string, value: unknown) {
	const [head, ...rest] = key.split('.') as [string, ...string[]];
	if (rest.length === 0) {
		if (value === undefined) delete out[head];
		else out[head] = value;
		return;
	}
	const sub = isTable(out[head]) ? copy(out[head]) : {};
	put(sub, rest.join('.'), value);
	if (Object.keys(sub).length) out[head] = sub;
	else delete out[head];
}

/** Form values for a selector: `all` for `{}`, `list` for the shorthand list. */
function selectorToForm(v: unknown, selects: 'host' | 'service'): Record<string, string> {
	const out: Record<string, string> = { all: '', list: '' };
	for (const [key] of SELECTOR_FIELDS[selects]) out[key] = '';
	if (Array.isArray(v)) {
		out.list = list(v).join(', ');
	} else if (isTable(v)) {
		if (Object.keys(v).length === 0) out.all = 'true';
		for (const [key] of SELECTOR_FIELDS[selects]) {
			const [a, b] = key.split('.');
			const value = b ? (isTable(v[a!]) ? (v[a!] as Record<string, unknown>)[b] : undefined) : v[a!];
			out[key] = key === 'expr' ? str(value) : list(value).join(', ');
		}
	}
	return out;
}

/** The selector to write; keys the form doesn't show are kept. */
export function selectorFromForm(form: Record<string, string>, raw: unknown, selects: 'host' | 'service'): unknown {
	if (form.list?.trim()) return items(form.list);
	const out: Record<string, unknown> = isTable(raw) ? copy(raw) : {};
	for (const [key] of SELECTOR_FIELDS[selects]) {
		const [a, b] = key.split('.');
		const text = (form[key] ?? '').trim();
		const value = key === 'expr' ? text || undefined : items(text).length ? items(text) : undefined;
		if (b) {
			const sub = isTable(out[a!]) ? (out[a!] as Record<string, unknown>) : {};
			if (value === undefined) delete sub[b];
			else sub[b] = value;
			if (Object.keys(sub).length) out[a!] = sub;
			else delete out[a!];
		} else if (value === undefined) delete out[a!];
		else out[a!] = value;
	}
	if (Object.keys(out).length === 0) return form.all === 'true' ? {} : undefined;
	return out;
}

/** Rows for the scalar vars of a table; table vars (for_each data) aren't shown. */
function varsToRows(v: unknown): [string, string][] {
	return Object.entries(isTable(v) ? v : {})
		.filter(([, x]) => !isTable(x))
		.map(([k, x]) => [k, str(x)] as [string, string]);
}

/** The vars to write: the rows, and the table vars of `before`. */
function varsFromRows(rows: [string, string][], before: unknown): Record<string, unknown> | undefined {
	const tables = Object.entries(isTable(before) ? before : {}).filter(([, x]) => isTable(x));
	const all = [...tables, ...rows.filter(([k]) => k.trim()).map(([k, x]) => [k.trim(), x] as [string, unknown])];
	return all.length === 0 ? undefined : Object.fromEntries(all);
}

function thresholdsToRows(v: unknown): [string, string, string][] {
	return Object.entries((v as Record<string, Record<string, unknown>>) ?? {}).map(
		([metric, t]) => [metric, str(t?.warn), str(t?.crit)] as [string, string, string]
	);
}

function thresholdsFromRows(rows: [string, string, string][]): Record<string, unknown> | undefined {
	const set = rows.filter(([m]) => m.trim());
	if (set.length === 0) return undefined;
	return Object.fromEntries(
		set.map(([m, warn, crit]) => {
			const t: Record<string, string> = {};
			if (warn.trim()) t.warn = warn.trim();
			if (crit.trim()) t.crit = crit.trim();
			return [m.trim(), t];
		})
	);
}

/** Keys of a host service entry the form shows. */
export const SERVICE_ROW_KEYS = ['enabled', 'interval', 'vars', 'thresholds'];

/** Form values for an object's fields (or for a new object). */
export function toForm(kind: Kind, raw: Record<string, unknown>): FormValues {
	const values: FormValues = {};
	for (const f of fields(kind)) {
		const v = value(raw, f);
		switch (f.type) {
			case 'list':
				values[f.key] = list(v).join(', ');
				break;
			case 'bool':
				values[f.key] = v === undefined ? (f.default ?? false) : v === true;
				break;
			case 'vars':
				values[f.key] = varsToRows(v);
				break;
			case 'thresholds':
				values[f.key] = thresholdsToRows(v);
				break;
			case 'selector':
				values[f.key] = selectorToForm(v, f.selects ?? 'host');
				break;
			case 'windows':
				values[f.key] = (Array.isArray(v) ? v : []).map((w) => {
					const t = isTable(w) ? w : {};
					return [list(t.days).join(', '), str(t.from), str(t.to)] as WindowRow;
				});
				break;
			case 'services':
				values[f.key] = Object.entries(isTable(v) ? v : {}).map(([name, e]): ServiceRow => {
					const entry = isTable(e) ? e : {};
					return {
						name,
						enabled: entry.enabled === false ? 'no' : entry.enabled === true ? 'yes' : '',
						interval: str(entry.interval),
						vars: varsToRows(entry.vars),
						thresholds: thresholdsToRows(entry.thresholds),
						orig: name,
						// Until the form knows what the host gets: keep it.
						own: true
					};
				});
				break;
			default:
				values[f.key] = str(v);
		}
	}
	return values;
}

function same(a: unknown, b: unknown) {
	return JSON.stringify(a) === JSON.stringify(b);
}

/**
 * The service entries to write; keys of an entry the form doesn't show are
 * kept. A row without any settings is left out, unless it is `own`.
 */
function servicesFromForm(rows: ServiceRow[], raw: unknown): unknown {
	const before = isTable(raw) ? raw : {};
	const out: Record<string, unknown> = {};
	for (const row of rows) {
		const name = row.name.trim();
		if (!name) continue;
		const old = isTable(before[row.orig]) ? before[row.orig] as Record<string, unknown> : {};
		const entry: Record<string, unknown> = copy(old);
		const set = (key: string, value: unknown) => {
			if (value === undefined || value === '') delete entry[key];
			else entry[key] = same(value, old[key]) ? old[key] : value;
		};
		set('enabled', row.enabled === 'no' ? false : row.enabled === 'yes' ? true : undefined);
		set('interval', row.interval.trim());
		set('vars', varsFromRows(row.vars, old.vars));
		set('thresholds', thresholdsFromRows(row.thresholds));
		if (Object.keys(entry).length > 0 || row.own) out[name] = entry;
	}
	return Object.keys(out).length ? out : undefined;
}

/**
 * The fields to write: `raw` with the form's keys replaced. Empty inputs
 * and default booleans remove the key; a value equal to what was written
 * keeps its original form (`groups = "x"` stays a string).
 */
export function fromForm(kind: Kind, raw: Record<string, unknown>, values: FormValues): Record<string, unknown> {
	const out: Record<string, unknown> = { ...raw };
	for (const f of fields(kind)) {
		const v = values[f.key];
		const before = value(raw, f);
		let next: unknown;
		switch (f.type) {
			case 'list': {
				const parts = items(String(v ?? ''));
				next = parts.length === 0 ? undefined : same(parts, list(before)) ? before : parts;
				break;
			}
			case 'bool':
				next = v === (f.default ?? false) ? undefined : v;
				// Keep an explicit default that was written.
				if (next === undefined && before === v) next = v;
				break;
			case 'int': {
				const text = String(v ?? '').trim();
				// Not a number: sent as text, so the server reports it.
				next = text === '' ? undefined : Number.isFinite(Number(text)) ? Number(text) : text;
				break;
			}
			case 'vars':
				// Table vars (for_each data) aren't in the form: kept.
				next = varsFromRows(v as [string, string][], before);
				break;
			case 'thresholds':
				next = thresholdsFromRows(v as [string, string, string][]);
				break;
			case 'selector':
				next = selectorFromForm(v as Record<string, string>, before, f.selects ?? 'host');
				break;
			case 'services':
				next = servicesFromForm(v as ServiceRow[], before);
				break;
			case 'windows': {
				const rows = (v as WindowRow[]).filter(([, from, to]) => from.trim() || to.trim());
				next =
					rows.length === 0
						? undefined
						: rows.map(([days, from, to]) => {
								const w: Record<string, unknown> = {};
								if (items(days).length) w.days = items(days);
								w.from = from.trim();
								w.to = to.trim();
								return w;
							});
				break;
			}
			default: {
				const text = String(v ?? '');
				next = text.trim() === '' ? undefined : f.type === 'mono' ? text : text.trim();
				// A number written in the file stays a number while unchanged.
				if (typeof before === 'number' && String(before) === next) next = before;
			}
		}
		put(out, f.key, next === undefined || !same(next, before) ? next : before);
	}
	// A short form stays short while only its value is set and unchanged.
	for (const f of fields(kind).filter((f) => f.short)) {
		const [head, key] = f.key.split('.') as [string, string];
		const t = out[head];
		if (typeof raw[head] === 'string' && isTable(t) && same(Object.keys(t), [key]) && t[key] === raw[head]) out[head] = raw[head];
	}
	return out;
}

/** Keys of `raw` the form doesn't show. */
export function otherKeys(kind: Kind, raw: Record<string, unknown>): string[] {
	const known = new Set(fields(kind).map((f) => f.key.split('.')[0]));
	return Object.keys(raw).filter((k) => !known.has(k));
}

/**
 * Names for clones: comma- or line-separated, with numeric ranges keeping
 * their padding (`web[03-05]` is web03, web04, web05). Throws on errors.
 */
export function expandNames(text: string, max = 1000): string[] {
	const out: string[] = [];
	for (const item of text.split(/[,\n]/).map((s) => s.trim()).filter(Boolean)) {
		const m = item.match(/^(.*)\[(\d+)-(\d+)\](.*)$/);
		if (!m) {
			out.push(item);
			continue;
		}
		const [, before, from, to, after] = m as unknown as [string, string, string, string, string];
		const [a, b] = [Number(from), Number(to)];
		if (b < a) throw new Error(`${item}: range goes down`);
		if (b - a + 1 + out.length > max) throw new Error(`more than ${max} names`);
		for (let n = a; n <= b; n++) out.push(`${before}${String(n).padStart(from.length, '0')}${after}`);
	}
	const dup = out.find((n, i) => out.indexOf(n) !== i);
	if (dup) throw new Error(`${dup} is listed twice`);
	if (out.length > max) throw new Error(`more than ${max} names`);
	return out;
}
