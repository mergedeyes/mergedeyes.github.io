// Types and calls for the hub API (/api/v1). Mirrors crates/hub/src/view.rs.

export type State = 'ok' | 'warning' | 'critical' | 'unknown';
export type HostState = 'up' | 'down';
export type StateType = 'soft' | 'hard';

export interface ServiceView {
	host: string;
	service: string;
	/** For display; null shows the name. */
	title: string | null;
	/** null while pending (no result yet) */
	state: State | null;
	state_type: StateType | null;
	attempt: number | null;
	max_attempts: number;
	last_check: string | null;
	last_change: string | null;
	output: string;
	passive: boolean;
	/** Show graphs of its metrics. */
	graphs: boolean;
	/** The state changes too often; problem notifications are held back. */
	flapping: boolean;
	/** Service groups. */
	groups: string[];
	acknowledged: boolean;
	/** Notifications off for it or its host. */
	muted: boolean;
	checks_enabled: boolean;
	/** In a downtime now. */
	downtime: boolean;
}

export interface Counts {
	ok: number;
	warning: number;
	critical: number;
	unknown: number;
	pending: number;
}

export interface HostView {
	name: string;
	address: string;
	node: string | null;
	/** For run_on = "host" services. */
	ssh: { user: string; port: number; secret: string } | null;
	/** Text, or tables (for_each data). */
	vars: Record<string, unknown>;
	/** Host groups. */
	groups: string[];
	enabled: boolean;
	/** From the host check; 'up' without one; null while the host check is pending. */
	state: HostState | null;
	state_type: StateType | null;
	attempt: number | null;
	max_attempts: number | null;
	last_check: string | null;
	last_change: string | null;
	output: string;
	flapping: boolean;
	/** Name of the host check. */
	check: string | null;
	/** The service whose graphs the host page shows. */
	graph: string | null;
	/** Worst service state; null if all are pending. */
	worst: State | null;
	services: Counts;
	/** The host problem is acknowledged. */
	acknowledged: boolean;
	muted: boolean;
	checks_enabled: boolean;
	downtime: boolean;
}

export interface HostCounts {
	up: number;
	down: number;
	pending: number;
}

/** Host and service totals; `generation` changes when states, flags or the config change. */
export interface Summary {
	generation: number;
	hosts: HostCounts & { total: number; unhandled: number };
	services: Counts & { total: number; unhandled: number };
}

/** Filters, sort and page of `GET /services`. */
export interface ServiceQuery {
	host?: string;
	hostgroup?: string;
	group?: string;
	/** Comma-separated states, `pending` included. */
	state?: string;
	problems?: boolean;
	hard?: boolean;
	unhandled?: boolean;
	q?: string;
	sort?: 'host' | 'service' | 'status' | 'last_check' | 'duration' | 'attempt';
	desc?: boolean;
	offset?: number;
	limit?: number;
}

/** Filters, sort and page of `GET /hosts`. */
export interface HostQuery {
	/** Comma-separated: up, down, pending. */
	state?: string;
	group?: string;
	unhandled?: boolean;
	q?: string;
	sort?: 'host' | 'status' | 'address' | 'last_check' | 'duration';
	desc?: boolean;
	offset?: number;
	limit?: number;
}

export interface ServiceList {
	/** Of the services that match every filter but `state`. */
	counts: Counts;
	/** Services that match every filter. */
	total: number;
	services: ServiceView[];
}

export interface HostList {
	counts: HostCounts;
	total: number;
	hosts: HostView[];
}

/** What a stream opened with `watch` gets results of. */
export interface Watch {
	/** `host/service` ids. */
	services?: string[];
	/** All services of these hosts. */
	hosts?: string[];
}

/** Acknowledgement, mute or disabled checks on a host or service. */
export interface Flag {
	kind: 'ack' | 'mute' | 'no_checks';
	/** `host` or `host/service`. */
	target: string;
	author: string;
	comment: string;
	created: string;
}

export interface Downtime {
	/** null: a window of a downtime group's schedule. */
	id: number | null;
	kind: 'host' | 'service' | 'hostgroup' | 'servicegroup' | 'downtimegroup';
	target: string;
	start: string;
	end: string;
	author: string | null;
	comment: string;
	/** Schedule entry, e.g. `sun 02:00-04:00`. */
	window: string | null;
	active: boolean;
	hosts: number;
	services: number;
}

export type Action = 'recheck' | 'ack' | 'unack' | 'mute' | 'unmute' | 'disable-checks' | 'enable-checks' | 'downtime';

/** What an action applies to, and its options. */
export interface ActionBody {
	hosts?: string[];
	/** `host/service` ids. */
	services?: string[];
	hostgroups?: string[];
	servicegroups?: string[];
	downtimegroups?: string[];
	comment?: string;
	start?: string;
	end?: string;
	duration?: string;
}

export interface ActionResult {
	action: Action;
	done: string[];
	skipped: { target: string; reason: string }[];
}

export interface ProcessInfo {
	version: string;
	pid: number;
	instance: string | null;
	started: string;
	uptime_s: number;
	config_loaded: string;
	database: string | null;
	notifications: boolean;
	checks: boolean;
	hosts: number;
	services: number;
	scheduler: { jobs: number; running: number; paused: number; passive: number };
	results: { last_1m: number; last_5m: number; last_15m: number };
	acknowledged: number;
	muted: number;
	checks_off: number;
	in_downtime: number;
}

export interface Scheduled {
	host: string;
	service: string;
	title: string | null;
	next_check: string | null;
	last_check: string | null;
	running: boolean;
	passive: boolean;
	paused: boolean;
	interval_s: number;
}

export interface Metric {
	name: string;
	value: number;
	unit?: string;
	labels?: Record<string, string>;
	/** evaluated thresholds */
	warn?: string;
	crit?: string;
	/** thresholds from the plugin output; Nagios ones are not evaluated */
	perf_warn?: string;
	perf_crit?: string;
	min?: number;
	max?: number;
	/** absent when the metric has no thresholds */
	state?: State;
}

export interface ServiceDetail extends ServiceView {
	last_hard_change: string | null;
	long_output: string;
	duration_ms: number | null;
	metrics: Metric[];
	/** Weighted state changes in the last 21 results, 0-100. */
	percent_change: number | null;
	/** Gives the host its UP/DOWN state. */
	host_check: boolean;
	check: {
		/** The check it runs; null for an inline command. */
		name: string | null;
		/** The service object it is made from (`disk` for the for_each service `disk_root`); null for a host's one-off service. */
		service: string | null;
		/** Why the host has it: `hostgroup linux`, `servicegroup basics on host web01`, ... */
		assigned: string[];
		command: string | null;
		interval_s: number;
		retry_interval_s: number;
		timeout_s: number;
		freshness_s: number | null;
		/** Flap detection thresholds in percent; null when off. */
		flap_low: number | null;
		flap_high: number | null;
		origin: string;
	};
	/** Downtimes it is in, e.g. `hostgroup linux`. */
	downtimes: string[];
	/** Set on it or its host. */
	flags: Flag[];
}

export interface Series {
	id: number;
	metric: string;
	labels: Record<string, string>;
	unit: string;
	/** Oldest and newest sample; null without samples. */
	first: string | null;
	last: string | null;
	/** Threshold bounds of the latest result, in `unit`: configured, else the plugin's. */
	warn: number[];
	crit: number[];
}

export interface Points {
	bucket: number;
	/** [unix ms, avg, min, max, count] */
	points: [number, number, number, number, number][];
}

/** How long after a result its history is in the database (ms): the hub writes in batches of about a second. */
export const HISTORY_DELAY = 1500;

/** `host`: UP/DOWN of the host, from its host check. */
export type EventKind = 'service' | 'host' | 'flap_start' | 'flap_stop' | 'ack' | 'downtime_start' | 'downtime_end';

/** An entry of the alert history. */
export interface StateEvent {
	host: string;
	service: string;
	ts: string;
	kind: EventKind;
	/** up/down for host events */
	from_state: State | HostState | null;
	to_state: State | HostState;
	state_type: StateType;
	hard: boolean;
	output: string;
	/** The service's title now. */
	title: string | null;
}

/** One notification to one contact over one channel (the notification log). */
export interface SentNotification {
	ts: string;
	host: string;
	/** For host notifications: the host check. */
	service: string;
	host_check: boolean;
	kind: 'problem' | 'recovery' | 'flappingstart' | 'flappingstop' | 'downtimestart' | 'downtimeend';
	state: State | HostState;
	/** A problem sent again: a repeat or an escalation. */
	repeat: boolean;
	since: string | null;
	contact: string;
	/** email, webhook, slack, ... or empty if the contact has none. */

export class ApiError extends Error {
	constructor(
		public status: number,
		message: string,
		/** The parsed JSON error body, for errors that carry details. */
		public body: Record<string, unknown> = {}
	) {
		super(message);
	}
	/** The hub runs without a database, so there is no history. */
	get noDatabase() {
		return this.status === 503 && this.message.includes('no database');
	}
}

/** Called on any 401: the session is missing or expired. */
let onUnauthorized: () => void = () => {};
export function setUnauthorizedHandler(handler: () => void) {
	onUnauthorized = handler;
}

async function get<T>(path: string, init?: RequestInit): Promise<T> {
	const res = await fetch(`/api/v1${path}`, init);
	if (!res.ok) {
		if (res.status === 401 && !path.startsWith('/auth/')) onUnauthorized();
		let message = `${res.status} ${res.statusText}`;
		let body = {};
		try {
			body = await res.json();
			message = (body as { error?: string }).error ?? message;
		} catch {
			// not JSON
		}
		throw new ApiError(res.status, message, body);
	}
	if (res.status === 204) return undefined as T;
	return res.json();
}

function send<T>(method: string, path: string, body?: unknown): Promise<T> {
	return get<T>(path, {
		method,
		headers: body === undefined ? {} : { 'content-type': 'application/json' },
		body: body === undefined ? undefined : JSON.stringify(body)
	});
}

function filePath(path: string) {
	return path.split('/').map(seg).join('/');
}

/** Query parameters, leaving out unset, empty and false values. */
function params(q: object): URLSearchParams {
	const out = new URLSearchParams();
	for (const [k, v] of Object.entries(q)) if (v !== undefined && v !== '' && v !== false) out.set(k, String(v));
	return out;
}
