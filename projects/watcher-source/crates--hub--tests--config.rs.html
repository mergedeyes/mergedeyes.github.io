//! Config editing over the API, with a stand-in for the run loop's reload.

use axum::body::Body;
use axum::http::{Request, StatusCode};
use http_body_util::BodyExt;
use serde_json::{Value, json};
use std::collections::HashMap;
use std::path::PathBuf;
use std::sync::Arc;
use tokio::sync::mpsc;
use tower::ServiceExt;
use watcher_config::{CheckDirs, ConfigMode, load_dir};
use watcher_hub::{AuthSettings, ConfigFiles, Hub, Reload};

const HOSTS: &str = r#"# Web servers
[[host]]
name = "web01"   # primary
groups = ["web"]

[check.http]
command = "check_http -H $host.address$"

[service.http]

[hostgroup.web]
services = ["http"]
"#;

struct Setup {
    hub: Arc<Hub>,
    dir: PathBuf,
    reloads: mpsc::UnboundedReceiver<String>,
}

fn setup(name: &str, mode: ConfigMode) -> Setup {
    let dir =
        std::env::temp_dir().join(format!("watcher-hub-config-{name}-{}", std::process::id()));
    let _ = std::fs::remove_dir_all(&dir);
    std::fs::create_dir_all(&dir).unwrap();
    std::fs::write(dir.join("hosts.toml"), HOSTS).unwrap();
    // check_http is a script in a check directory.
    let checks = dir.with_extension("checks");
    let _ = std::fs::remove_dir_all(&checks);
    std::fs::create_dir_all(&checks).unwrap();
    std::fs::write(checks.join("check_http"), "#!/bin/sh\necho OK\n").unwrap();
    std::fs::set_permissions(
        checks.join("check_http"),
        std::os::unix::fs::PermissionsExt::from_mode(0o755),
    )
    .unwrap();
    let config = Arc::new(load_dir(&dir).unwrap().config);
    let (control, _control_rx) = mpsc::channel(8);
    let hub = Hub::new(config, HashMap::new(), control, None, AuthSettings::OFF);

    // Reload like the run loop does, and report who asked.
    let (reload_tx, mut reload_rx) = mpsc::channel::<Reload>(4);
    let (seen_tx, reloads) = mpsc::unbounded_channel();
    let (h, d) = (hub.clone(), dir.clone());
    tokio::spawn(async move {
        while let Some(request) = reload_rx.recv().await {
            h.set_config(Arc::new(load_dir(&d).unwrap().config));
            let _ = seen_tx.send(request.by);
            let _ = request.reply.send(Ok(()));
        }
    });
    hub.enable_config_editing(ConfigFiles {
        dir: dir.clone(),
        mode,
        reload: reload_tx,
        check_dirs: CheckDirs::new(vec![checks]),
    });
    Setup { hub, dir, reloads }
}

async fn call(hub: &Arc<Hub>, method: &str, uri: &str, body: Option<Value>) -> (StatusCode, Value) {
    let mut req = Request::builder().method(method).uri(uri);
    let body = match body {
        Some(json) => {
            req = req.header("content-type", "application/json");
            Body::from(json.to_string())
        }
        None => Body::empty(),
    };
    let resp = hub
        .clone()
        .router()
        .oneshot(req.body(body).unwrap())
        .await
        .unwrap();
    let status = resp.status();
    let bytes = resp.into_body().collect().await.unwrap().to_bytes();
    (
        status,
        serde_json::from_slice(&bytes).unwrap_or(Value::Null),
    )
}

#[tokio::test]
async fn lists_files_and_objects() {
    let s = setup("list", ConfigMode::Hybrid);
    let (status, overview) = call(&s.hub, "GET", "/api/v1/config", None).await;
    assert_eq!(status, StatusCode::OK);
    assert_eq!(overview["mode"], "hybrid");
    assert_eq!(overview["writable"], true);
    assert_eq!(overview["files"][0]["path"], "hosts.toml");
    assert_eq!(overview["diagnostics"], json!([]));

    let (_, file) = call(&s.hub, "GET", "/api/v1/config/files/hosts.toml", None).await;
    assert_eq!(file["text"], HOSTS);
    assert_eq!(file["hash"], overview["files"][0]["hash"]);
    let (status, _) = call(&s.hub, "GET", "/api/v1/config/files/../etc.toml", None).await;
    assert!(status.is_client_error());

    let (_, objects) = call(&s.hub, "GET", "/api/v1/config/objects", None).await;
    let host = &objects["objects"][0];
    assert_eq!(host["kind"], "host");
    assert_eq!(host["name"], "web01");
    assert_eq!(host["line"], 2);
    assert_eq!(host["fields"]["groups"], json!(["web"]));

    let (_, scripts) = call(&s.hub, "GET", "/api/v1/config/scripts", None).await;
    assert_eq!(scripts["scripts"][0]["name"], "check_http");
}

#[tokio::test]
async fn create_preview_apply() {
    let mut s = setup("apply", ConfigMode::Hybrid);
    let (status, change) = call(
        &s.hub,
        "POST",
        "/api/v1/config/edit",
        Some(json!({ "kind": "host", "fields": { "name": "web02", "groups": ["web"] } })),
    )
    .await;
    assert_eq!(status, StatusCode::OK, "{change}");
    assert_eq!(change["path"], "ui/hosts.toml");
    assert_eq!(change["base"], Value::Null);

    let body = json!({ "changes": [change] });
    let (_, preview) = call(&s.hub, "POST", "/api/v1/config/preview", Some(body.clone())).await;
    assert_eq!(preview["valid"], true);
    assert_eq!(preview["changes"]["hosts"]["added"], json!(["web02"]));
    assert_eq!(
        preview["changes"]["services"]["added"],
        json!(["web02/http"])
    );
    assert!(!s.dir.join("ui/hosts.toml").exists());

    let (status, applied) = call(&s.hub, "POST", "/api/v1/config/apply", Some(body.clone())).await;
    assert_eq!(status, StatusCode::OK, "{applied}");
    assert_eq!(applied["changes"]["hosts"]["added"], json!(["web02"]));
    assert_eq!(s.reloads.recv().await.unwrap(), "local user");
    assert_eq!(s.hub.config().services.len(), 2);
    assert!(
        std::fs::read_to_string(s.dir.join("ui/hosts.toml"))
            .unwrap()
            .contains("web02")
    );

    // The same change again: the file exists now.
    let (status, conflict) = call(&s.hub, "POST", "/api/v1/config/apply", Some(body)).await;
    assert_eq!(status, StatusCode::CONFLICT);
    assert_eq!(conflict["path"], "ui/hosts.toml");
    assert!(
        conflict["current"]["text"]
            .as_str()
            .unwrap()
            .contains("web02")
    );
}

#[tokio::test]
async fn invalid_changes_are_not_written() {
    let s = setup("invalid", ConfigMode::Hybrid);
    let (_, file) = call(&s.hub, "GET", "/api/v1/config/files/hosts.toml", None).await;
    let broken = HOSTS.replace("check_http", "check_http $host.nope$");
    let body =
        json!({ "changes": [{ "path": "hosts.toml", "base": file["hash"], "text": broken }] });

    let (_, preview) = call(&s.hub, "POST", "/api/v1/config/preview", Some(body.clone())).await;
    assert_eq!(preview["valid"], false);
    let problem = &preview["diagnostics"][0];
    assert_eq!(problem["path"], "hosts.toml");
    assert_eq!(problem["line"], 6);
    assert_eq!(problem["severity"], "error");

    let (status, _) = call(&s.hub, "POST", "/api/v1/config/apply", Some(body)).await;
    assert_eq!(status, StatusCode::UNPROCESSABLE_ENTITY);
    assert_eq!(
        std::fs::read_to_string(s.dir.join("hosts.toml")).unwrap(),
        HOSTS
    );

    // Deleting the host would leave the host group without hosts: a warning only.
    let (_, change) = call(
        &s.hub,
        "POST",
        "/api/v1/config/edit",
        Some(json!({ "kind": "host", "name": "web01" })),
    )
    .await;
    let (_, preview) = call(
        &s.hub,
        "POST",
        "/api/v1/config/preview",
        Some(json!({ "changes": [change] })),
    )
    .await;
    assert_eq!(preview["valid"], true);
    assert_eq!(
        preview["changes"]["services"]["removed"],
        json!(["web01/http"])
    );
    assert_eq!(preview["diagnostics"][0]["severity"], "warning");
}

#[tokio::test]

    let s = setup("clones", ConfigMode::Hybrid);

    .await;
    assert_eq!(status, StatusCode::OK, "{body}");
    let changes = body["changes"].as_array().unwrap();
    assert_eq!(changes.len(), 1);
    assert_eq!(changes[0]["path"], "hosts.toml");
    let (_, preview) = call(
        &s.hub,
        "POST",
        "/api/v1/config/preview",
        Some(json!({ "changes": changes })),
    )
    .await;
    assert_eq!(
        preview["changes"]["hosts"]["added"],
        json!(["web02", "web03"])
    );
    assert_eq!(
        preview["changes"]["services"]["added"],
        json!(["web02/http", "web03/http"])
    );

    // A clone with a name that exists: 409.

        &s.hub,

#[tokio::test]
async fn objects_in_one_step() {
    let mut s = setup("onestep", ConfigMode::Hybrid);
    let (status, applied) = call(
        &s.hub,
        "PUT",
        "/api/v1/config/objects/host/web02",
        Some(json!({ "address": "10.0.0.6", "groups": ["web"] })),
    )
    .await;
    assert_eq!(status, StatusCode::OK, "{applied}");
    // An address: the default ping host check.
    assert_eq!(
        applied["changes"]["services"]["added"],
        json!(["web02/http", "web02/ping"])
    );
    assert_eq!(s.reloads.recv().await.unwrap(), "local user");
    let (_, host) = call(&s.hub, "GET", "/api/v1/config/objects/host/web02", None).await;
    assert_eq!(host["path"], "ui/hosts.toml");
    assert_eq!(host["fields"]["address"], "10.0.0.6");

    // Update in place, then an invalid change is refused.
    let (status, _) = call(
        &s.hub,
        "PUT",
        "/api/v1/config/objects/host/web02",
        Some(json!({ "address": "10.0.0.7", "groups": ["web"] })),
    )
    .await;
    assert_eq!(status, StatusCode::OK);
    let (status, err) = call(
        &s.hub,
        "PUT",
        "/api/v1/config/objects/host/web02",
        Some(json!({ "groups": ["nope"] })),
    )
    .await;
    assert_eq!(status, StatusCode::UNPROCESSABLE_ENTITY, "{err}");
    let (_, objects) = call(&s.hub, "GET", "/api/v1/config/objects?kind=host", None).await;
    assert_eq!(objects["objects"].as_array().unwrap().len(), 2);

    let (status, applied) = call(&s.hub, "DELETE", "/api/v1/config/objects/host/web02", None).await;
    assert_eq!(status, StatusCode::OK, "{applied}");
    assert_eq!(applied["changes"]["hosts"]["removed"], json!(["web02"]));
    let (status, _) = call(&s.hub, "GET", "/api/v1/config/objects/host/web02", None).await;
    assert_eq!(status, StatusCode::NOT_FOUND);

    let (status, _) = call(&s.hub, "POST", "/api/v1/config/reload", None).await;
    assert_eq!(status, StatusCode::OK);
}

async fn call_leased(
    hub: &Arc<Hub>,
    method: &str,
    uri: &str,
    lease: &str,
    body: Value,
) -> StatusCode {
    let req = Request::builder()
        .method(method)
        .uri(uri)
        .header("content-type", "application/json")
        .header("x-watcher-lease", lease)
        .body(Body::from(body.to_string()))
        .unwrap();
    hub.clone().router().oneshot(req).await.unwrap().status()
}

#[tokio::test]
async fn edit_locks() {
    let s = setup("locks", ConfigMode::Hybrid);
    let lock = "/api/v1/config/locks/hosts.toml";
    let (status, a) = call(&s.hub, "PUT", lock, Some(json!({}))).await;
    assert_eq!(status, StatusCode::OK);
    let a = a["lease"].as_str().unwrap().to_string();
    assert!(s.dir.join(".hosts.toml.watcher-lock").exists());

    // Someone else: refused, with who has it.
    let (status, held) = call(&s.hub, "PUT", lock, Some(json!({}))).await;
    assert_eq!(status, StatusCode::LOCKED);
    assert_eq!(held["holder"]["via"], "web");
    assert_eq!(
        held["error"],
        "local user is editing hosts.toml in the web UI"
    );
    let (_, file) = call(&s.hub, "GET", "/api/v1/config/files/hosts.toml", None).await;
    let change = json!({ "changes": [{
        "path": "hosts.toml",
        "base": file["hash"],
        "text": format!("{HOSTS}\n# edited\n"),
    }] });
    let (status, _) = call(&s.hub, "POST", "/api/v1/config/apply", Some(change.clone())).await;
    assert_eq!(status, StatusCode::LOCKED);
    let (status, _) = call(&s.hub, "DELETE", "/api/v1/config/objects/host/web01", None).await;
    assert_eq!(status, StatusCode::LOCKED);
    // The holder writes.
    let status = call_leased(&s.hub, "POST", "/api/v1/config/apply", &a, change).await;
    assert_eq!(status, StatusCode::OK);

    // Renewing keeps the lease; taking over ends it.
    let (status, _) = call(&s.hub, "PUT", lock, Some(json!({ "lease": a }))).await;
    assert_eq!(status, StatusCode::OK);
    let (status, b) = call(&s.hub, "PUT", lock, Some(json!({ "force": true }))).await;
    assert_eq!(status, StatusCode::OK);
    let b = b["lease"].as_str().unwrap().to_string();
    let (status, _) = call(&s.hub, "PUT", lock, Some(json!({ "lease": a }))).await;
    assert_eq!(status, StatusCode::LOCKED);
    let (_, locks) = call(&s.hub, "GET", "/api/v1/config/locks", None).await;
    assert_eq!(locks["locks"].as_array().unwrap().len(), 1);
    let (_, overview) = call(&s.hub, "GET", "/api/v1/config", None).await;
    assert_eq!(overview["locks"][0]["path"], "hosts.toml");

    let (status, _) = call(&s.hub, "DELETE", &format!("{lock}?lease={b}"), None).await;
    assert_eq!(status, StatusCode::NO_CONTENT);
    assert!(!s.dir.join(".hosts.toml.watcher-lock").exists());
    let (_, locks) = call(&s.hub, "GET", "/api/v1/config/locks", None).await;
    assert_eq!(locks["locks"], json!([]));
}

#[tokio::test]
async fn files_open_in_an_editor() {
    let s = setup("editor", ConfigMode::Hybrid);
    // A vim swap file of a running process (this one).
    let mut swap = vec![0u8; 1024];
    swap[..9].copy_from_slice(b"b0VIM 9.1");
    swap[24..28].copy_from_slice(&std::process::id().to_le_bytes());
    swap[28..32].copy_from_slice(b"anna");
    let host = watcher_config::locks::local_host().unwrap_or_default();
    swap[68..68 + host.len()].copy_from_slice(host.as_bytes());
    std::fs::write(s.dir.join(".hosts.toml.swp"), swap).unwrap();

    let (_, locks) = call(&s.hub, "GET", "/api/v1/config/locks", None).await;
    assert_eq!(locks["locks"][0]["via"], "vim");
    assert_eq!(locks["locks"][0]["by"], "anna");
    let lock = "/api/v1/config/locks/hosts.toml";
    let (status, held) = call(&s.hub, "PUT", lock, Some(json!({}))).await;
    assert_eq!(status, StatusCode::LOCKED);
    assert_eq!(held["error"], "anna is editing hosts.toml in vim");
    let (status, _) = call(&s.hub, "DELETE", "/api/v1/config/objects/host/web01", None).await;
    assert_eq!(status, StatusCode::LOCKED);
    // Taken over: writes with the lease go through, vim is still reported.
    let (status, taken) = call(&s.hub, "PUT", lock, Some(json!({ "force": true }))).await;
    assert_eq!(status, StatusCode::OK);
    assert_eq!(taken["others"][0]["via"], "vim");
    let req = Request::builder()
        .method("DELETE")
        .uri("/api/v1/config/objects/host/web01")
        .header("x-watcher-lease", taken["lease"].as_str().unwrap())
        .body(Body::empty())
        .unwrap();
    let status = s.hub.clone().router().oneshot(req).await.unwrap().status();
    assert_eq!(status, StatusCode::OK);
}
