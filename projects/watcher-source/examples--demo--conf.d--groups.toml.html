members = ["*"]
services = ["ping"]
host_check = "ping"

[hostgroup.linux]
title = "Linux Servers"
services = ["load"]

[hostgroup.web-servers]
title = "Web Servers"
services = ["http"]

[hostgroup.db-servers]
title = "Database Servers"
servicegroups = ["postgres"]

[hostgroup.mq]
title = "Message Queues"
members = ["mq-*"]
services = ["queue_depth"]

[hostgroup.ibmi]
title = "IBM i"
servicegroups = ["ibmi"]

[hostgroup.dc1]
title = "Datacenter 1"
members = ["web*", "db*"]

[servicegroup.postgres]
title = "PostgreSQL"
services = ["replication", "pg_connections"]

[servicegroup.ibmi]
title = "IBM i"
services = ["qsysopr_messages", "asp_usage"]

[servicegroup.webshop]
title = "Webshop"

# No notifications for Linux servers on Sunday nights.
[downtimegroup.patchday]
title = "Patch Day"
hostgroups = ["linux"]
