<script lang="ts">
	import { goto } from '$app/navigation';
	import { page } from '$app/state';
	import { untrack } from 'svelte';
	import { api, type Change, type Kind } from '#lib/api.ts';
	import ApplyDialog from '#lib/ApplyDialog.svelte';
	import { config, fileHref, objectId } from '#lib/config.svelte.ts';
	import ConfigHead from '#lib/ConfigHead.svelte';
	import Combobox from '#lib/Combobox.svelte';
	import ConfigTree from '#lib/ConfigTree.svelte';
	import HostServices from '#lib/HostServices.svelte';
	import { FileLease } from '#lib/lease.svelte.ts';
	import LockNotice from '#lib/LockNotice.svelte';
	import { live } from '#lib/live.svelte.ts';
	import { Suggest } from '#lib/suggest.ts';
	import MatchPreview from '#lib/MatchPreview.svelte';
	import { RELATIONS, remote, without } from '#lib/membership.ts';
	import { KINDS, SCHEMAS, SELECTOR_FIELDS, expandNames, fromForm, otherKeys, selectorFromForm, toForm, type Field, type FormValues, type Group, type ServiceRow, type WindowRow } from '#lib/schema.ts';

	const WEEKDAYS = ['mon', 'tue', 'wed', 'thu', 'fri', 'sat', 'sun'];
	/** Settings of commands, which built-in checks don't have. */
	const COMMAND_ONLY = ['shell', 'run_on', 'copy', 'passive'];
	const TIMES = ['00:00', '01:00', '02:00', '03:00', '04:00', '05:00', '06:00', '12:00', '18:00', '20:00', '22:00', '23:00', '24:00'];

	let kind = $derived(KINDS.find((k) => k.path === page.params.kind));
	let name = $derived(page.url.searchParams.get('name'));
	let object = $derived(kind && name !== null ? config.find(kind.kind, name) : undefined);
	/** Clone mode (`?clone=x`): a new form filled from x, saved into x's file. */
	let cloneName = $derived(name === null ? page.url.searchParams.get('clone') : null);
	/** A service to highlight in the services table (`?focus=http`, from a service page). */
	let focus = $derived(page.url.searchParams.get('focus'));
	let source = $derived(kind && cloneName !== null ? config.find(kind.kind, cloneName) : undefined);
	let raw = $derived.by<Record<string, unknown>>(() => {
		if (object) return object.fields;
		if (!source) return {};
		const { name: _, ...rest } = source.fields;
		return rest;
	});

	let values = $state<FormValues | null>(null);
	let pending = $state<Change[] | null>(null);
	let deleting = $state(false);
	let error = $state<string | null>(null);

	$effect(() => {
		config.load();
	});

	/** The edit lease on the object's file; editing an existing object needs it. */
	let lease = $state<FileLease | null>(null);
	let leasePath = $derived(config.writable ? object?.path : undefined);
	$effect(() => {
		if (!leasePath) {
			lease = null;
			return;
		}
		// Once free, the object may have changed: fill the form again.
		const p = leasePath;
		const l = untrack(
			() =>
				new FileLease(p, () => {
					config.load().then(() => (values = null));
				})
		);
		lease = l;
		return () => l.stop();
	});
	let editable = $derived(config.writable && (!object || lease?.held === true));

	let scripts = $state<{ dirs: string[]; scripts: { name: string; dir: string }[] } | null>(null);
	/** Names of the services on hosts. */
	let serviceNames = $state<string[]>([]);
	$effect(() => {
		void live.reloads;
		api.names().then((n) => (serviceNames = n.services), () => (serviceNames = []));
	});
	/** What `graph` can name, as Watcher compiled the saved host or group: its services with graphs. */
	let graphNames = $state<string[]>([]);
	$effect(() => {
		void live.reloads;
		const k = kind?.kind;
		if (name === null || (k !== 'host' && k !== 'hostgroup')) {
			graphNames = [];
			return;
		}
		api.names({ [k]: name, graphs: true }).then((n) => (graphNames = n.services), () => (graphNames = []));
	});
	$effect(() => {
		if (kind && SCHEMAS[kind.kind].some((g) => g.fields.some((f) => f.scripts)))
			api.configScripts().then((s) => (scripts = s), () => (scripts = null));
	});

	// Another object: start over.
	$effect(() => {
		void kind;
		void name;
		void cloneName;
		untrack(() => {
			values = null;
			dropped = [];
			const type = page.url.searchParams.get('type');
			added = type ? [type] : [];
			removed = [];
		});
	});

	/** Extra sections (Proxmox VE) added with their button or `?type=`, and those taken out. */
	let added = $state<string[]>([]);
	let removed = $state<string[]>([]);

	/** Whether a section shows: always, or for an extra one, when set in the file or added. */
	function shows(g: Group): boolean {
		if (!g.extra) return true;
		if (removed.includes(g.extra)) return false;
		return added.includes(g.extra) || g.fields.some((f) => raw[f.key.split('.')[0]!] !== undefined);
	}

	/** Takes an extra section out: its fields are cleared, so saving removes them. */
	function removeExtra(g: Group) {
		for (const f of g.fields) values![f.key] = f.type === 'bool' ? (f.default ?? false) : '';
		removed = [...removed, g.extra!];
		added = added.filter((x) => x !== g.extra);
	}

	// Fill the form once the object is there; later reloads don't reset edits.
	$effect(() => {
		if (!kind || !config.loaded) return;
		if (cloneName !== null && !source) return;
		const fields = raw;
		const k = kind.kind;
		untrack(() => {
			if (values !== null) return;
			const filled = toForm(k, fields);
			// "New Service" from a built-in check in the check list.
			const preset = page.url.searchParams.get('check');
			if (k === 'service' && name === null && cloneName === null && preset) filled.check = preset;
			// From a service page: the row of that service, opened.
			const rows = filled.service as ServiceRow[] | undefined;
			if (focus && k === 'host' && rows && !rows.some((r) => r.name === focus)) rows.push(serviceRow(focus));
			values = filled;
			rowsSorted = false;
			requestAnimationFrame(() => document.querySelector('.item.focused')?.scrollIntoView({ block: 'center' }));
		});
	});

	let others = $derived(kind ? otherKeys(kind.kind, raw) : []);

	/** Group membership kept on the other side (a host group listing this host, ...). */
	let relation = $derived(kind ? RELATIONS[kind.kind] : undefined);
	let otherPath = $derived(KINDS.find((k) => k.kind === relation?.other)?.path ?? '');
	let remotes = $derived.by(() => {
		if (!kind || !relation || !values || !object) return [];
		return remote(config.objects, kind.kind, object.name, items(String(values[relation.field] ?? '')));
	});
	/** Other objects this form takes the membership out of on save. */
	let dropped = $state<string[]>([]);

	/** Edits of the other side: `drop` loses this object, the rest follow a rename. */
	function remoteEdits(drop: string[], newName: string | null) {
		if (!relation || !object) return [];
		const rel = relation;
		const old = object.name;
		return remotes
			.filter((r) => r.listed && (drop.includes(r.name) || (newName !== null && newName !== old)))
			.flatMap((r) => {
				const o = config.find(rel.other, r.name);
				if (!o) return [];
				const fields = without(o, rel.otherField, old);
				if (!drop.includes(r.name) && newName) fields[rel.otherField] = [...list(fields[rel.otherField]), newName];
				return [{ kind: rel.other, name: r.name, fields }];
			});
	}

	function list(v: unknown): string[] {
		return v === undefined ? [] : Array.isArray(v) ? v.map(String) : [String(v)];
	}

	/** Selectors for the match preview of a field, or null for none. */
	function previewOf(f: Field): { kind: 'host' | 'service'; selectors: unknown[]; none?: string } | null {
		if (!kind || !values) return null;
		if (f.type === 'selector') {
			const selects = f.selects ?? 'host';
			const sel = selectorFromForm(values[f.key] as Record<string, string>, raw[f.key], selects);
			return { kind: selects, selectors: sel === undefined ? [] : [sel], none: selects === 'host' ? 'All hosts' : 'All services' };
		}
		const names = (key: string) => items(String(values![key] ?? ''));
		if (kind.kind === 'hostgroup' && f.key === 'members') {
			return { kind: 'host', selectors: names('members').length ? [names('members')] : [] };
		}
		if (kind.kind === 'downtimegroup' && f.key === 'hostgroups') {
			const sels: unknown[] = [];
			if (names('members').length) sels.push(names('members'));
			if (names('hostgroups').length) sels.push({ groups: names('hostgroups') });
			return { kind: 'host', selectors: sels, none: 'No hosts' };
		}
		return null;
	}

	let suggest = $derived(
		new Suggest(config.objects, serviceNames, scripts?.scripts ?? [], config.builtins.map((b) => b.name), graphNames)
	);

	/** The values of a select field; built-in checks come from the server. */
	function choices(f: Field): string[] {
		return f.key === 'builtin' ? config.builtins.map((b) => b.name) : (f.options ?? []);
	}

	/** Greyed out: a field that rules this one out has a value (check or command, ...). */
	function blocked(f: Field): boolean {
		// Built-in checks run inside Watcher: no shell, no SSH, not passive.
		if (builtin && COMMAND_ONLY.includes(f.key)) return true;
		return (f.exclusive ?? []).some((k) => String(values?.[k] ?? '').trim() !== '');
	}

	/** The check a service runs: its own, or the one named like it. */
	let serviceCheck = $derived(
		kind?.kind === 'service' && values && !String(values.command ?? '').trim()
			? String(values.check ?? '').trim() || String(values.name ?? '').trim()
			: ''
	);

	/** The built-in check this form sets up, whose options get fields of their own. */
	let builtin = $derived.by(() => {
		if (!kind || !values) return undefined;
		if (kind.kind === 'check') return config.builtins.find((b) => b.name === String(values!.builtin ?? ''));
		return serviceCheck ? config.builtinOf(serviceCheck) : undefined;
	});

	/** What the service's check object sets, which applies without a value here. */
	let inherited = $derived.by(() => {
		const v = serviceCheck ? config.find('check', serviceCheck)?.fields.vars : undefined;
		if (typeof v !== 'object' || v === null || Array.isArray(v)) return {} as Record<string, string>;
		return Object.fromEntries(Object.entries(v).map(([k, x]) => [k, typeof x === 'object' ? '' : String(x)]));
	});

	/** Options with fields of their own; tables stay in Variables. */
	function ownField(name: string): boolean {
		return builtin?.vars.some((v) => v.name === name && v.type !== 'table') ?? false;
	}

	function option(name: string): string {
		return rows<[string, string]>('vars').find((r) => r[0] === name)?.[1] ?? '';
	}

	function setOption(name: string, value: string) {
		if (!values) return;
		const list = rows<[string, string]>('vars');
		const i = list.findIndex((r) => r[0] === name);
		if (value === '') values.vars = list.filter((_, j) => j !== i);
		else if (i >= 0) list[i]![1] = value;
		else values.vars = [...list, [name, value]];
	}

	/** What applies to an option left empty. */
	function fallback(name: string, shown: string | null): string {
		const from = inherited[name];
		if (from) return `${from} (from check ${serviceCheck})`;
		return shown ?? '';
	}

	/** Services the host gets (as written), by name: where from. Host services with these names change them on this host. */
	let assignedServices = $derived.by(() => {
		const out = new Map<string, string>();
		if (kind?.kind !== 'host' || !values) return out;
		const host = String(values.name ?? '');
		const listOf = (v: unknown) => (Array.isArray(v) ? v.map(String) : typeof v === 'string' ? [v] : []);
		const groupServices = (sg: string) => [
			...listOf(config.find('servicegroup', sg)?.fields.services),
			...config.of('service').filter((o) => listOf(o.fields.groups).includes(sg)).map((o) => o.name)
		];
		const add = (names: string[], from: string) => names.forEach((n) => out.has(n) || out.set(n, from));
		add(items(String(values.services ?? '')), 'host');
		for (const sg of items(String(values.servicegroups ?? ''))) add(groupServices(sg), `servicegroup ${sg}`);
		const listed = items(String(values.groups ?? ''));
		for (const g of config.of('hostgroup')) {
			const member = listed.includes(g.name) || listOf(g.fields.members).some((m) => glob(m).test(host));
			if (!member) continue;
			add(listOf(g.fields.services), `hostgroup ${g.name}`);
			for (const sg of listOf(g.fields.servicegroups)) add(groupServices(sg), `servicegroup ${sg} on hostgroup ${g.name}`);
		}
		// The host check: its own, a host group's, or ping for a host with an address.
		const fromGroups = config
			.of('hostgroup')
			.filter((g) => listed.includes(g.name) || listOf(g.fields.members).some((m) => glob(m).test(host)))
			.map((g) => g.fields.host_check)
			.find((hc) => hc !== undefined);
		const own = String(values.host_check ?? '').trim();
		const hc = own !== '' ? own : fromGroups !== undefined ? String(fromGroups) : String(values.address ?? '').trim() ? 'ping' : '';
		if (hc !== '' && hc !== 'false') add([hc], 'host check');
		return out;
	});

	/** An empty row for service `name`. */
	function serviceRow(name: string): ServiceRow {
		return { name, enabled: '', interval: '', vars: [], thresholds: [], orig: '', own: false };
	}

	/** The `for_each` service a row belongs to (`disk` for `disk_root`), if the host gets it. */
	function forEachBase(name: string): string | undefined {
		return [...assignedServices.keys()].find((s) => name.startsWith(`${s}_`) && config.find('service', s)?.fields.for_each !== undefined);
	}

	/** Set once the rows from the file are sorted into the host's services. */
	let rowsSorted = false;

	// The rows follow what the host gets: one per service. A service it no
	// longer gets loses its row and settings; an entry that alone puts a
	// service on the host (`own`) stays until removed.
	$effect(() => {
		if (kind?.kind !== 'host' || !values) return;
		const assigned = assignedServices;
		untrack(() => {
			const current = (values!.service as ServiceRow[] | undefined) ?? [];
			if (!rowsSorted) {
				for (const r of current) if (r.orig !== '' && (assigned.has(r.name) || forEachBase(r.name))) r.own = false;
				rowsSorted = true;
			}
			const next = current.filter((r) => r.own || assigned.has(r.name) || forEachBase(r.name) !== undefined);
			for (const name of assigned.keys()) if (!next.some((r) => r.name === name)) next.push(serviceRow(name));
			const order = [...assigned.keys()];
			const rank = (r: ServiceRow) => {
				const i = order.indexOf(forEachBase(r.name) ?? r.name);
				return i < 0 ? order.length : i;
			};
			next.sort((a, b) => rank(a) - rank(b) || a.name.localeCompare(b.name));
			if (next.length !== current.length || next.some((r, i) => r !== current[i])) values!.service = next;
		});
	});

	/** Where a row comes from, for the table. */
	function rowFrom(row: ServiceRow): string {
		const base = forEachBase(row.name);
		if (base) return `for_each item of ${base}`;
		return assignedServices.get(row.name) ?? 'this host only';
	}

	/** The interval a service has without one of its own on this host. */
	function serviceInterval(name: string): string {
		const svc = config.find('service', forEachBase(name) ?? name)?.fields;
		const check = config.find('check', String(svc?.check ?? forEachBase(name) ?? name))?.fields;
		return String(svc?.interval ?? check?.interval ?? '1m');
	}

	function glob(pattern: string): RegExp {
		const re = pattern.replace(/[.+^${}()|[\]\\]/g, '\\$&').replace(/\*/g, '.*').replace(/\?/g, '.');
		return new RegExp(`^${re}$`);
	}

		object ? (config.overview?.diagnostics ?? []).filter((p) => p.path === object.path && p.line === object.line) : []
	);
	let missing = $derived(config.loaded && ((name !== null && !object) || (cloneName !== null && !source)));

	async function save(event: SubmitEvent) {
		event.preventDefault();
		if (!kind || !values) return;
		error = null;
		try {
			const fields = fromForm(kind.kind, raw, values);
			const extra = remoteEdits(dropped, typeof fields.name === 'string' ? fields.name : null);
			if (source) {
				pending = await clones(kind.kind, fields, source.path);
			} else if (extra.length) {
				pending = (await api.editMany([{ kind: kind.kind, name, fields }, ...extra])).changes;
			} else {
				pending = [await api.editObject(kind.kind, name, fields)];
			}
			deleting = false;
		} catch (e) {
			error = e instanceof Error ? e.message : String(e);
		}
	}

	async function remove() {
		if (!kind || name === null) return;
		error = null;
		try {
			// Lists on the other side lose it too.
			const extra = remoteEdits(remotes.map((r) => r.name), null);
			pending = extra.length
				? (await api.editMany([{ kind: kind.kind, name, fields: null }, ...extra])).changes
				: [await api.editObject(kind.kind, name, null)];
			deleting = true;
		} catch (e) {
			error = e instanceof Error ? e.message : String(e);
		}
	}

	/** One clone per name, into the file of the original. Hosts cloned several times get their address from the name. */
	async function clones(k: Kind, fields: Record<string, unknown>, path: string): Promise<Change[]> {
		const list = expandNames(String(fields.name ?? ''));
		const edits =
			list.length === 0
				? [{ kind: k, name: null, fields, path }]
				: list.map((n) => {
						const f: Record<string, unknown> = { ...fields, name: n };
						if (k === 'host' && list.length > 1) delete f.address;
						return { kind: k, name: null, fields: f, path };
					});
		return (await api.editMany(edits)).changes;
	}

	function closed(applied: boolean) {
		pending = null;
		if (applied && kind) goto(`/config/${kind.path}`);
	}

	function items(text: string) {
		return text
			.split(',')
			.map((s) => s.trim())
			.filter(Boolean);
	}

	/** In clone mode, the name field takes several names. */
	function shown(f: Field): Field {
		if (f.key !== 'name' || cloneName === null) return f;
		return { ...f, label: 'Names', placeholder: 'web03, web[04-10]', help: 'one clone per name; hosts cloned several times: address = name' };
	}

	function rows<T>(key: string): T[] {
		return (values?.[key] as T[]) ?? [];
	}
</script>
